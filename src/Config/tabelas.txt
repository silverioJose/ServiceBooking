Tabela Serviços

CREATE TABLE `saas`.`servicos` (`id` INT NOT NULL AUTO_INCREMENT , `nome` VARCHAR(100) NOT NULL , `descricao` TEXT NOT NULL , `duracao_minutos` INT NOT NULL , `preco` DECIMAL(10,2) NOT NULL , `foto` VARCHAR(255) NOT NULL , PRIMARY KEY (`id`)) ENGINE = InnoDB;

Tabela Profissionais

CREATE TABLE `saas`.`profissionais` (
    `id` INT NOT NULL AUTO_INCREMENT,
    `nome` VARCHAR(100) NOT NULL,
    `email` VARCHAR(100) NULL,
    `telefone` VARCHAR(20) NULL,
    `foto` VARCHAR(255) NULL,
    `ativo` TINYINT DEFAULT 1,
    PRIMARY KEY (`id`)
) ENGINE = InnoDB;


Tabela Agendamentos

CREATE TABLE `saas`.`agendamentos` (
    `id` INT NOT NULL AUTO_INCREMENT,
    `servico_id` INT NOT NULL,
    `profissional_id` INT NOT NULL,
    `data` DATE NOT NULL,
    `hora` TIME NOT NULL,
    `nome_cliente` VARCHAR(100) NOT NULL,
    `telefone_cliente` VARCHAR(20) NOT NULL,
    `status` VARCHAR(20) DEFAULT 'confirmado',
    `criado_em` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    FOREIGN KEY (`servico_id`) REFERENCES `saas`.`servicos`(`id`),
    FOREIGN KEY (`profissional_id`) REFERENCES `saas`.`profissionais`(`id`)
) ENGINE = InnoDB;
